  ElMessageBox.confirm(`确定要删除备份 ${backup.name} 吗？`, '提示', {
    confirmButtonText: '确定',
    cancelButtonText: '取消',
    type: 'warning'
  }).then(() => {
    const index = backups.value.findIndex(b => b.name === backup.name)
    if (index > -1) {
      backups.value.splice(index, 1)
      ElMessage.success('备份已删除')
    }
  }).catch(() => {})
}

  try {
    const data = await getSettings()
    settings.value = data || []
  } catch { settings.value = [] }
})
